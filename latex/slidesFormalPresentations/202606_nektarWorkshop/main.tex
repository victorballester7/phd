\begin{frame}
	\frametitle{First questions}
	\begin{itemize}
		\item Small disturbances in the boundary layer initially grow as $\propto \exp{n(x)}$.
		\item We aim to understand how much a gap of width $w$ and depth $d$ can overamplify these disturbances.
		\item For each gap of width $w$ and depth $d$, we define $$\Delta n_{\text{gap}} := n_\text{gap} - n_\text{flat}$$
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Conclusions}
	\begin{itemize}
		\item Very good agreement between the numerical predictions (2d inc.) and the experimental (3d low Mach wind tunnel) results.
		\item The 3D instability doesn't seem to be strong enough to destabilise the downstream boundary layer. Only after the 2D instability sets in, the transition location advances till the downstream edge of the gap.
		\item For deep gaps, Mach number has a destabilising effect, while for shallow gaps, higher Mach numbers lead to later transition.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{A temporary solution: Jacobian free approach}

	Navier-Stokes equations can be written in the form
	$$
		\partial_t \mathbf{q} = \mathbf{N}(\mathbf{q})
	$$

	Assume we want to linearize $\mathbf{N}$ around a base flow $\mathbf{\overline{q}}$ so that we have:
	$$
		\partial_t \mathbf{q'} = \mathbf{L}({\mathbf{\overline{q}}}) \mathbf{q'} \quad \text{with} \quad \mathbf{L}({\mathbf{\overline{q}}}) = \frac{\partial \mathbf{N}}{\partial \mathbf{q}} ({\mathbf{\overline{q}}})
	$$
	We can write evolution operators for both the non-linear and linearized systems as
	$$
		\mathbf{q}_t = \vf{\mathcal{A}}_t^\text{nonlin} \mathbf{q}_0 \quad \text{and} \quad \mathbf{q}_t' = \vf{\mathcal{A}}_t^\text{lin} (\mathbf{\overline{q}})\mathbf{q}_0'
	$$
	and we can relate them as
	$$
		\vf{\mathcal{A}}_t^\text{lin}(\mathbf{\overline{q}}) = \pdv{\vf{\mathcal{A}}_t^\text{nonlin}}{\mathbf{q}_0}(\mathbf{\overline{q}})
	$$

\end{frame}

\begin{frame}
	If we seek disturbances of the form
	$$
		\mathbf{q'}(\mathbf{x},t)
		=
		\hat{\mathbf q}(\mathbf{x}) e^{\lambda t},
	$$
	the linearized problem becomes
	$$
		\mathbf L(\overline{\mathbf q}) \hat{\mathbf q}
		=
		\lambda \hat{\mathbf q}.
	$$

	We are usually interested in the eigenvalues of
	$\mathbf L(\bar{\mathbf q})$
	with largest real part (Arnoldi algorithm),
	which are equivalent to the eigenvalues of the
	evolution operator
	$$
		\vf{\mathcal A}_t^{\text{lin}}
		=
		\exp{{\mathbf L(\overline{\mathbf q}) t}}
	$$
	with largest magnitude (Modified Arnoldi algorithm).

\end{frame}
\begin{frame}

	Using the relation
	$$
		\vf{\mathcal{A}}_t^{\text{lin}}(\mathbf{\overline{q}}) = \pdv{\vf{\mathcal{A}}_t^\text{nonlin}}{\mathbf{q}_0}(\mathbf{\overline{q}})
	$$
	we can compute the evolution of the linear system as a finite-difference approximation
	$$
	\vf{\mathcal{A}}_t^{\text{lin}}(\mathbf{\overline{q}})\mathbf v
		\approx
		\frac{
			\vf{\mathcal{A}}_t^{\text {nonlin}}
		(\bar{\mathbf q}+\varepsilon \mathbf v)
		-
		\vf{\mathcal{A}}_t^{\text{nonlin}}
		(\bar{\mathbf q})
		}{\varepsilon},
	$$
	where $\varepsilon=\|\vf{\overline{q}}\|_2\sqrt{\epsilon_\text{machine}}$ is chosen in a similar fashion to the one we use for the nonlinear solve of Compressible NS eqs.

	\textbf{Num. of computations:}
	\begin{itemize}
		\item 1st instability: If $\overline{\mathbf q}$ is an ``accuarate" fixed point of $\vf{\mathcal{A}}_t^{\rm nonlin}(\bar{\mathbf q})$ then  $\vf{\mathcal{A}}_t^{\rm nonlin}(\bar{\mathbf q})\approx \bar{\mathbf q}$. $\implies$ $N_\text{iterArnoldi}$ evaluations of $\vf{\mathcal{A}}_t^{\rm nonlin}$.
		\item 2nd instability: If $\overline{\mathbf q}$ is a limit cycle, then we need to evaluate $\vf{\mathcal{A}}_t^{\rm nonlin}(\bar{\mathbf q})$ once at the beginning of the algorithm and store the result for the rest of the iterations. $\implies$ $N_\text{iterArnoldi} + 1$ evaluations of $\vf{\mathcal{A}}_t^{\rm nonlin}$.
	\end{itemize}

\end{frame}
