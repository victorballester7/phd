% \draw[domain=-3.5:-1, samples=100, smooth,variable=\x, orange, ultra thick] plot ({\x}, {0.5*exp(-(\x+2.6)^2*5)*cos((\x+2.6)*180/3.14159)*cos(20*(\x+2.6)*180/3.14159)});

% Parameters for Gaussian forcing
\pgfmathsetmacro{\xx}{-3.01}   % center x
\pgfmathsetmacro{\yy}{0.49}    % center y
\pgfmathsetmacro{\gw}{10.0}     % Gaussian width
\pgfmathsetmacro{\k}{8.0}     % wave number for cos

% Forcing region drawn with orange dots
\begin{scope}
  \foreach \x in {-3.5,-3.39,...,-2.52} {
    \foreach \y in {0.0,0.11,...,0.98} {
      % Gaussian envelope
      \pgfmathsetmacro{\gauss}{exp(-\gw*((\x-\xx)^2+(\y-\yy)^2))}
      % Oscillation
      \pgfmathsetmacro{\osc}{cos(\k*(\x-\xx))}
      % Forcing amplitude
      \pgfmathsetmacro{\val}{\gauss*\osc}
      % Only draw if amplitude is positive (optional)
      \ifdim \val pt > 0pt
					\fill[orange,opacity=1] (\x,\y) circle[radius={0.04*\val}];
      \fi
    }
  }
\end{scope}

% Wall
\draw[ultra thick] (\aa,0) -- (0,0) -- (0,-\dd) -- (\ww,-\dd) -- (\ww,0) -- (\bb,0);

\def\lengthWall{0.5}

}
\foreach \x in {0,0.5,...,3} {
    \draw[thick] (\x,-\dd) -- ++(225:\lengthWall); % Wall hatching
}
